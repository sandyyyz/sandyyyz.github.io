struct linux_binprm ​

该结构体用于存放在载入二进制文件时，需要保存的必要信息。

c

/*
 * This structure is used to hold the arguments that are used when loading binaries.
 */
struct linux_binprm {
#ifdef CONFIG_MMU
        struct vm_area_struct *vma;
        unsigned long vma_pages;
        unsigned long argmin; /* rlimit marker for copy_strings() */
#else
# define MAX_ARG_PAGES  32
        struct page *page[MAX_ARG_PAGES];
#endif
        struct mm_struct *mm;
        unsigned long p; /* current top of mem */
        unsigned int
                /* Should an execfd be passed to userspace? */
                have_execfd:1,

                /* Use the creds of a script (see binfmt_misc) */
                execfd_creds:1,
                /*
                 * Set by bprm_creds_for_exec hook to indicate a
                 * privilege-gaining exec has happened. Used to set
                 * AT_SECURE auxv for glibc.
                 */
                secureexec:1,
                /*
                 * Set when errors can no longer be returned to the
                 * original userspace.
                 */
                point_of_no_return:1,
                /* Set when "comm" must come from the dentry. */
                comm_from_dentry:1,
                /*
                 * Set by user space to check executability according to the
                 * caller's environment.
                 */
                is_check:1;
        struct file *executable; /* Executable to pass to the interpreter */
        struct file *interpreter;
        struct file *file;
        struct cred *cred;      /* new credentials */
        int unsafe;             /* how unsafe this exec is (mask of LSM_UNSAFE_*) */
        unsigned int per_clear; /* bits to clear in current->personality */
        int argc, envc;
        const char *filename;   /* Name of binary as seen by procps */
        const char *interp;     /* Name of the binary really executed. Most
                                   of the time same as filename, but could be
                                   different for binfmt_{misc,script} */
        const char *fdpath;     /* generated filename for execveat */
        unsigned interp_flags;
        int execfd;             /* File descriptor of the executable */
        unsigned long exec;

        struct rlimit rlim_stack; /* Saved RLIMIT_STACK used during exec. */

        char buf[BINPRM_BUF_SIZE];
} __randomize_layout;

int bprm_execve(struct linux_binprm *bprm) ​

  1. 为当前任务准备creds
  2. check unsafe exec
  3. 标记current->in_execve = 1
  4. 停用当前任务的mm_cid并释放其占用的cid资源
  5. 调用sched_exec()， 为当前任务重新选择CPU，在必要时触发一次迁移
  6. security_bprm_creds_for_exec, 通过LSM钩子准备并检查新进程的安全凭证
  7. 接下来调用 exec_binprm
  8. sched_mm_cid_after_execve(current)
  9. rseq_execve(current)
  10. current->in_execve = 0
  11. user_events_execve(current)
  12. acct_update_integrals(current)
  13. task_numa_free(current, false)
  14. return.

※ 补注：sched_exec 选在 exec 这个时机迁移任务，因为 exec 即将重建 cache 局部性，此刻迁移成本最低（fs/exec.c:1840）。

static int exec_binprm(struct linux_binprm *bprm) ​

  1. save current->pid, and read old_vpid.
  2. looping, search_binary_handler(bprm),which will call fmt->load_binary(bprm) inside.
  3. audit_bprm(bprm), save record to audit if audit_context exists.
  4. trace_sched_process_exec(current, old_pid, bprm)
  5. ptrace_event(PTRACE_EVENT_EXEC, old_vpid)
  6. proc_exec_connector(current)
  7. return.

※ 补注：binfmt "套娃"——脚本等格式加载失败后由下一层 handler 接管， bprm->file = bprm->interpreter 交接（fs/exec.c:1801），循环最多 5 层 防 -ELOOP（fs/exec.c:1788）。

  1. ptrace_event(PTRACE_EVENT_EXEC, old_vpid)
  2. proc_exec_connector(current)
  3. return.

static int load_elf_binary ​

步骤总览：

  1. check elf magic number
  2. load elf program headers
  3. try to find interpreter in program headers, get it's struct file and elf_header if exists,PT_INTERP
  4. process special segments of the elf binary: PT_GNU_STACK and PT_LOPROC/HIPROC
  5. check interpreter's legality and get its program headers,load_elf_interpreter
  6. process special segments of the interpreter PT_GNU_STACK and PT_LOPROC/HIPROC.
  7. parse_elf_properties(解析 ELF 文件中 PT_GNU_PROPERTY 段里的 GNU 属性记录
  8. arch_check_elf,(dummy implementation)
  9. begin_new_exec
  10. set personality（见文末"疑问"）
  11. set randomize va space if needed.
  12. setup_new_exec(bprm), arch_pick_mmap_layout, pick mmap layout for new mm according to personality and rlimit, then arch_setup_new_exec, do arch-related setting.
  13. setup_arg_pages, determine the final position and the permission of new stack, and expand stack space.
  14. map the ELF image into the correct location in memory, with each loadable segment's addr = load_bias + elf_ppnt->p_vaddr
  15. load the interpreter into the correct location in memory.load_elf_interp
  16. allow_write_access for interpreter, exe_file_allow_write_access(interpreter)
  17. free interpreter's loaded elf header and program headers table.
  18. free program's program headers table.
  19. set_binfmt(&elf_format), set current->mm->binfmt = &elf_format.
  20. setup additional pages if arch has.ARCH_SETUP_ADDITIONAL_PAGES(bprm, elf_ex, !!interpreter)
  21. create_elf_table(bprm, elf_ex, interp_load_addr, e_entry, phdr_addr)
  22. elf_coredump_set_mm_eflags(mm, elf_ex->e_flags)
  23. for static PIE, move the brk area out of the mmap region and into the unused ELF_ET_DYN_BASE region.
  24. pagealign the elf_brk.
  25. randomize the brk if config PF_RANDOMIZE && snapshot_randomize_va_space > 1.
  26. simulate svr4's behavior(maps page0 as read-only then do_mseal) if needed.
  27. initialize partial registers to zero according to the platform-related abi.
  28. store any stack rlimit changes before starting thread by function finalize_exec(bprm)
  29. start new thread,START_THREAD(elf_ex, regs, elf_entry, bprm->p)
  30. return retval=0.

Do the simple consistency checks for the interpreter, as the elf above, which is :

  • Check ELFMAG
  • Check arch correctness
  • Check interperter's program headers's PT.. as same as above
c

	/* Some simple consistency checks for the interpreter */
	if (interpreter) {
		retval = -ELIBBAD;
		/* Not an ELF interpreter */
		if (memcmp(interp_elf_ex->e_ident, ELFMAG, SELFMAG) != 0)
			goto out_free_dentry;
		/* Verify the interpreter has a valid arch */
		if (!elf_check_arch(interp_elf_ex) ||
		    elf_check_fdpic(interp_elf_ex))

						   interpreter);

		elf_property_phdata = NULL;

			case PT_GNU_PROPERTY:
				elf_property_phdata = elf_ppnt;
				break;

			case PT_LOPROC ... PT_HIPROC:

							  elf_ppnt, interpreter,
							  true, &arch_state);
				if (retval)
					goto out_free_dentry;
				break;

static int create_elf_tables ​

  1. p = arch_align_stack(p), 初始用户栈指针做随机对齐，减少l1 cache 争用
  2. if the architecture has a platform capbility string, copy it to userspace.
  3. copy 16 random bytes to user for userpace PRNG seeding.
  4. create elf interpreter info(aux vectors) in mm->saved_auxv(in the kernel). NEW_AUX_ENT(XXXX)
  5. put argc on the stack
  6. put all argv on the stack ended by 0
  7. put all list of envp pointers on the stack, ended by 0
  8. put aux vectors on the stack, ended by AUX_NULL(0)
  9. return 0

向下增长架构（CONFIG_STACK_GROWSUP 未定义）的最终布局，从高地址到低地址：

高地址
+--------------------------------------+
| argv/envp 字符串区                    |  <- mm->arg_start ... mm->env_end
| （由 copy_strings 提前放置）          |
| bprm->exec 指向的 execfn 字符串       |
+--------------------------------------+

static int exec_mmap(struct mm_struct *mm) ​

  1. Notify parent that we're no longer interested in the old VM, exec_mm_release, 清理tsk和mm关联所产生的副作用，不是释放mm内存！
    1. clean futex resources, futex_exit_release
    2. clean mm-related resources? mm_release
      1. free uprobe
      2. decativate any cached register state
      3. announce clear_child_tid
      4. wake up vfork parent process
  2. acquire write lock of &tsk->signal->exec_update_lock, down_write_killable(&tsk->signal_exec_update_lock)
  3. acquire read lock of &old_mm->mmap_lock,mmap_read_lock_killable(old_mm)
  4. acquire &tsk->alloc_lock, task_lock(tsk)
  5. set &bprm->mm->membarrier_state = 0, runqueues.membarrier_state=0, membarrier_exec_mmap(mm)
  6. change tsk->mm and tsk->active_mm = mm
  7. mm_init_cid
  8. enable/disable irq
  9. add new mm to lru list
  10. find a new owner of old_mm
  11. mmput(old_mm)
  12. mmdrop_lazy_tlb(active_mm)
